"""Domain-level value object representing the generic filter tree language."""
from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, Optional

FilterDict = Dict[str, Any]


@dataclass(frozen=True)
class SearchFilterTree:
    """Immutable wrapper around the tree-based filter dictionary.

    The internal ``raw`` attribute must follow the grammar understood by
    :class:`application.processors.filter_builder.FilterBuilder`. The grammar is
    intentionally flexible but follows a few simple rules:

    * Logical nodes use the keys ``and``, ``or`` or ``not`` and contain nested
      clauses. Example::

          {
              "and": [
                  {"status": "ATIVO"},
                  {
                      "or": [
                          {"has_bdr": True},
                          {"market": {"in": ["NM"]}}
                      ]
                  }
              ]
          }

    * Leaf nodes describe conditions about a single column. They can be direct
      comparisons (``{"field": 10}``), dictionaries with operators such as
      ``{"field": {">=": 0}}`` or text/list operations like ``contains`` and
      ``in``.

    The goal of this value object is to provide a stable contract that can be
    shared between CLI, FastAPI adapters and the frontend while keeping the
    domain protected from ad-hoc dictionaries.
    """

    raw: FilterDict

    @classmethod
    def from_raw(cls, data: Optional[Any]) -> Optional["SearchFilterTree"]:
        """Coerce ``data`` into a :class:`SearchFilterTree` when possible."""
        if data is None:
            return None
        if isinstance(data, SearchFilterTree):
            return data
        if not isinstance(data, dict):
            raise TypeError(
                "SearchFilterTree.from_raw expects a dict or SearchFilterTree, "
                f"received {type(data)}"
            )
        return cls(raw=data)

    def to_dict(self) -> FilterDict:
        """Expose the underlying filter dictionary."""
        return self.raw

    def is_empty(self) -> bool:
        """Return ``True`` when the wrapped tree has no conditions."""
        return not bool(self.raw)

from __future__ import annotations

from typing import Any, Dict, Optional

from pydantic import BaseModel, Field

from domain.value_objects import SearchFilterTree


class SearchFiltersDTO(BaseModel):
    """DTO that mirrors the tree structure accepted by ``FilterBuilder``.

    The ``tree`` attribute can be omitted (``null``) when no filters are
    required. When provided, it must follow the same grammar described in
    :class:`domain.value_objects.search_filter_tree.SearchFilterTree`.
    """

    tree: Optional[Dict[str, Any]] = Field(
        default=None,
        description="Filter tree compatible with the backend SearchFilterTree contract.",
    )

    def to_domain(self) -> Optional[SearchFilterTree]:
        """Convert this DTO into the domain value object."""
        return SearchFilterTree.from_raw(self.tree)

from __future__ import annotations

from typing import Any, Dict

FilterDict = Dict[str, Any]


def build_ratios_filter_by_issuing_company(issuing_company: str) -> FilterDict:
    """Return a simple filter tree that matches the given issuing company."""
    company = issuing_company.strip()
    if not company:
        raise ValueError("issuing_company must be a non-empty string")

    return {
        "and": [
            {
                "issuing_company": {
                    "==": company,
                    "case": False,
                }
            }
        ]
    }
