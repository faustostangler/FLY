from __future__ import annotations

from dataclasses import dataclass
from datetime import date
from typing import List

from application.dtos.account_series_dto import AccountSeriesDTO, AccountSeriesPointDTO
from application.ports.uow_port import UowFactoryPort
from domain.ports.repository_account_series_port import RepositoryAccountSeriesPort
from domain.value_objects.account_code import AccountCode


@dataclass(slots=True)
class GetAccountSeriesUseCase:
    """Obtém série temporal de uma conta contábil para um ticker."""

    repository: RepositoryAccountSeriesPort
    uow_factory: UowFactoryPort

    def __call__(
        self,
        *,
        ticker: str,
        account_code: str,
        start_date: date | None = None,
        end_date: date | None = None,
    ) -> AccountSeriesDTO:
        account = AccountCode(account_code)

        with self.uow_factory() as uow:
            points = self.repository.get_account_series(
                ticker=ticker,
                account=account,
                start_date=start_date,
                end_date=end_date,
            )
            uow.commit()

        dto_points: List[AccountSeriesPointDTO] = [
            AccountSeriesPointDTO(date=point.date, value=float(point.value))
            for point in points
        ]
        dto_points.sort(key=lambda item: item.date)

        label = f"Conta {account.value}"

        return AccountSeriesDTO(
            ticker=ticker,
            account_code=account.value,
            label=label,
            points=dto_points,
        )

from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True)
class AccountCode:
    """Value object que representa o código contábil utilizado nas séries."""

    value: str

    def __post_init__(self) -> None:
        v = self.value.strip()
        if not v:
            raise ValueError("AccountCode vazio")

        parts = v.split(".")
        if any(not part.isdigit() for part in parts):
            raise ValueError(f"AccountCode inválido: {v}")

        object.__setattr__(self, "value", v)

    def __str__(self) -> str:
        return self.value

from __future__ import annotations

from datetime import date
from typing import Callable, Iterable, List

from sqlalchemy import select
from sqlalchemy.orm import Session

from application.ports.config_port import ConfigPort
from application.ports.logger_port import LoggerPort
from domain.entities.account_series import AccountPoint
from domain.ports.repository_account_series_port import RepositoryAccountSeriesPort
from domain.value_objects.account_code import AccountCode
from infrastructure.adapters.engine_setup import EngineSetup
from infrastructure.models import BaseModel
from infrastructure.models.financial_statement_model import FinancialStatementModel


class RepositoryAccountSeriesSqlAlchemy(EngineSetup, RepositoryAccountSeriesPort):
    """Implementação SQLAlchemy do repositório de séries contábeis."""

    def __init__(self, *, config: ConfigPort, logger: LoggerPort) -> None:
        super().__init__(
            config.database.connection_string,
            logger,
            orm_base=BaseModel,
        )
        self._config = config
        self._logger = logger

    def get_account_series(
        self,
        *,
        ticker: str,
        account: AccountCode,
        start_date: date | None = None,
        end_date: date | None = None,
    ) -> List[AccountPoint]:
        stmt = (
            select(FinancialStatementModel)
            .where(FinancialStatementModel.ticker == ticker)
            .where(FinancialStatementModel.account_code == account.value)
        )
        if start_date is not None:
            stmt = stmt.where(FinancialStatementModel.reference_date >= start_date)
        if end_date is not None:
            stmt = stmt.where(FinancialStatementModel.reference_date <= end_date)

        stmt = stmt.order_by(FinancialStatementModel.reference_date.asc())

        with self.Session() as session:  # type: Session
            rows: Iterable[FinancialStatementModel] = session.execute(stmt).scalars().all()

        return [
            AccountPoint(date=row.reference_date, value=float(row.value))
            for row in rows
        ]

    @property
    def session_factory(self) -> Callable[[], Session]:
        """Expose the SQLAlchemy session factory for composition roots."""

        return self.Session

from presentation.backend.dto.chart_dto import ChartDTO, PlotlyTraceDTO

router = APIRouter(
    prefix="/api/charts/accounts",
    tags=["charts"],
)


@router.get("/line/{ticker}", response_model=ChartDTO)
async def get_account_line_chart(
    ticker: str,
    account_code: str = Query(..., description="Código da conta (ex.: 02.03)"),
    start_date: date | None = Query(default=None),
    end_date: date | None = Query(default=None),
    usecase: GetAccountSeriesUseCase = Depends(get_account_series_usecase),
) -> ChartDTO:
    series = usecase(
        ticker=ticker,
        account_code=account_code,
        start_date=start_date,
        end_date=end_date,
    )

    if not series.points:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Série temporal vazia para esta conta",
        )

    trace = PlotlyTraceDTO(
        x=[point.date for point in series.points],
        y=[point.value for point in series.points],
        name=f"{series.account_code} – {series.label}",
        type="scatter",
        mode="lines",
        extra={
            "hovertemplate": "%{x}<br>Valor: %{y}<extra></extra>",
        },
    )

    layout = {
        "title": f"{series.label} – {series.ticker}",
        "xaxis": {"title": "Data"},
        "yaxis": {"title": "Valor"},
    }

    return ChartDTO(data=[trace], layout=layout)

import axios from 'axios'

const api = axios.create({
  baseURL: 'http://localhost:8000',
})

export async function fetchChart(params) {
  const payload = params || {}
  const type = String(payload.type || '').trim()
  if (!type) {
    throw new Error('Ticker inválido para carregar o gráfico')
  }
  const selection = Array.isArray(payload.selection) ? payload.selection : []

  const query = {}
  if (selection.length) {
    query.selection = selection.join(',')
  }

  const res = await api.get(`/charts/${encodeURIComponent(type)}`, {
    params: query,
  })
  return res.data
}

export async function fetchAccountLineChart(ticker, accountCode, params = {}) {
  const normalizedTicker = String(ticker || '').trim()
  const normalizedAccount = String(accountCode || '').trim()

  if (!normalizedTicker || !normalizedAccount) {
    throw new Error('Ticker e código da conta são obrigatórios')
  }

  const res = await api.get(
    `/api/charts/accounts/line/${encodeURIComponent(normalizedTicker)}`,
    {
      params: {
        account_code: normalizedAccount,
        ...params,
      },
    }
  )
  return res.data
}

export async function searchCompanies(filterQuery) {
  const res = await api.post('/companies/search', filterQuery || { clauses: [] })
  return res.data
}

export async function fetchCompanyFacets() {
  const res = await api.get('/companies/facets')
  return res.data
}

import { defineStore } from 'pinia'
import { fetchAccountLineChart } from '../services/apiService'

const DEFAULT_ACCOUNTS = ['02.03', '03.01']

function createStateFromAccounts(accounts) {
  return accounts.reduce(
    (acc, code) => {
      acc.charts[code] = null
      acc.isLoading[code] = false
      acc.error[code] = ''
      return acc
    },
    { charts: {}, isLoading: {}, error: {} }
  )
}

export const useAccountChartsStore = defineStore('accountCharts', {
  state: () => ({
    ticker: '',
    ...createStateFromAccounts(DEFAULT_ACCOUNTS),
  }),

  actions: {
    setTicker(ticker) {
      this.ticker = String(ticker || '').trim()
    },

    async loadAccountChart(accountCode, params = {}) {
      const code = String(accountCode || '').trim()
      if (!code) {
        return
      }

      if (!this.ticker) {
        this.error[code] = 'Ticker e código da conta são obrigatórios'
        return
      }

      this.isLoading[code] = true
      this.error[code] = ''

      try {
        const chart = await fetchAccountLineChart(this.ticker, code, params)
        this.charts[code] = chart
      } catch (err) {
        console.error(err)
        this.error[code] = err?.message || 'Falha ao carregar gráfico'
        this.charts[code] = null
      } finally {
        this.isLoading[code] = false
      }
    },

    async loadMultipleAccounts(accountCodes, params = {}) {
      const codes = (accountCodes || [])
        .map((code) => String(code || '').trim())
        .filter(Boolean)

      await Promise.all(
        codes.map((code) => this.loadAccountChart(code, params[code] || {}))
      )
    },
  },
})
