from __future__ import annotations

import json
from dataclasses import dataclass
from typing import Any, Dict, Optional


FilterDict = Dict[str, Any]


def _canonical_json(data: FilterDict) -> str:
    """Return a canonical JSON representation used for hashing."""
    return json.dumps(data, sort_keys=True, ensure_ascii=False, separators=(",", ":"))


@dataclass(frozen=True)
class SearchFilterTree:
    """Value object that encapsulates the structured filter tree used across the app."""

    raw: FilterDict

    @classmethod
    def from_raw(cls, data: Optional[Any]) -> Optional["SearchFilterTree"]:
        """Coerce a raw dict (or already wrapped instance) into ``SearchFilterTree``."""
        if data is None:
            return None
        if isinstance(data, SearchFilterTree):
            return data
        if not isinstance(data, dict):
            raise TypeError(
                "SearchFilterTree expects a mapping compatible with the FilterBuilder grammar."
            )
        return cls(raw=data)

    def to_dict(self) -> FilterDict:
        """Expose the raw tree as a plain dictionary."""
        return self.raw

    def is_empty(self) -> bool:
        """True when no condition is declared inside the tree."""
        return not bool(self.raw)

    def to_hash(self) -> str:
        """Return a deterministic hash-friendly representation."""
        if self.is_empty():
            return "empty"
        return _canonical_json(self.raw)

from __future__ import annotations

from typing import Any, Dict

FilterDict = Dict[str, Any]


def build_ratios_filter_by_issuing_company(issuing_company: str) -> FilterDict:
    """Retorna um filtro simples para restringir ratios por empresa emissora."""
    company = (issuing_company or "").strip()
    if not company:
        raise ValueError("issuing_company não pode ser vazio")

    return {
        "and": [
            {
                "issuing_company": {
                    "==": company,
                    "case": False,
                }
            }
        ]
    }


def build_default_ratios_filter() -> FilterDict:
    """Filtro padrão reutilizado pelo CLI ao processar ratios."""
    return {
        "and": [
            {"industry_segment": {"ne": "TELECOMUNICACOES", "case": False}},
        ]
    }
